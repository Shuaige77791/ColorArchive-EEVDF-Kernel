# pick_eevdf — earliest eligible virtual deadline:
+static struct sched_entity *pick_eevdf(struct cfs_rq *cfs_rq)
+{
+	struct rb_node *node = cfs_rq->tasks_timeline.rb_root.rb_node;
+	struct sched_entity *se = __pick_first_entity(cfs_rq);
+	struct sched_entity *curr = cfs_rq->curr;
+
+	if (curr && curr->on_rq) {
+		if (se && entity_before(curr, se))
+			return curr;
+		if (entity_eligible(cfs_rq, curr))
+			return curr;
+	}
+
+	/* RB-tree traversal for earliest deadline */
+	while (node) {
+		struct sched_entity *right = rb_entry(node, struct sched_entity, run_node);
+		if (!entity_eligible(cfs_rq, right)) {
+			node = node->rb_left;
+			continue;
+		}
+		se = right;
+		node = node->rb_right;
+	}
+	return se;
+}
